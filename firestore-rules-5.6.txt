rules_version = '2';

// FitPot Firestore security rules.
// Paste into Firebase console → Firestore Database → Rules → Publish.
//
// Data model
//   users/{uid}                          name, photoURL, groupId, photoConsent
//   passcodes/{code}                     groupId            (lookup for joining)
//   groups/{gid}                         name, passcode, penalty, currency, weeklyTarget, photoRequired, timezone, createdBy, createdAt
//   users/{uid}                          + groupIds [all your groups], groupId (the one you're viewing)  (phase 5.2), leftGroupIds (5.3)
//   groups/{gid}/members/{uid}           name, photoURL, passcode, timezone (own phone's), joinedAt, joinedDate, leftAt? (phase 5.2)
//   groups/{gid}/checkins/{date_uid}     uid, date, createdAt
//   groups/{gid}/restdays/{date}         votes {uid: time}
//   groups/{gid}/penalties/{id}          penalty, effectiveDate, setBy, setAt
//   groups/{gid}/photos/{date_uid}       uid, date, fileId (Google Drive), createdAt, slot, shot, activity?
//                                        up to 4 a day: {date_uid}, {date_uid}_2, _3, _4 (phase 5.6)
//   groups/{gid}/targets/{id}            target (days/week), effectiveDate (a Monday), setBy, setAt
//   groups/{gid}/tzchanges/{id}          uid, timezone, effectiveDate, setAt          (phase 5.1)
//   groups/{gid}/currencies/{id}         from, currency, rate, penalty, effectiveDate, setBy, setAt (phase 5.1)
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    // Joining or creating a group needs a confirmed email (Google accounts always are).
    function verified() { return request.auth.token.email_verified == true; }
    function isMember(groupId) {
      return signedIn() &&
        exists(/databases/$(database)/documents/groups/$(groupId)/members/$(request.auth.uid));
    }
    // A member who hasn't pressed Leave.
    function isActiveMember(groupId) {
      return isMember(groupId) &&
        get(/databases/$(database)/documents/groups/$(groupId)/members/$(request.auth.uid)).data.get('leftAt', null) == null;
    }
    function validCurrency(c) { return c in ['INR', 'USD', 'GBP', 'CAD']; }
    function validTz(tz) { return tz is string && tz.size() > 0 && tz.size() < 64; }

    // Profiles: any signed-in user can read names/avatars; only you write yours.
    match /users/{uid} {
      allow read: if signedIn();
      allow write: if signedIn() && request.auth.uid == uid;
    }

    // Passcode lookup: you can fetch one code if you know it, but never list them all.
    match /passcodes/{code} {
      allow get: if signedIn();
      allow list: if false;
      // Created in the same batch as its group, only by the group's creator.
      allow create: if signedIn()
        && getAfter(/databases/$(database)/documents/groups/$(request.resource.data.groupId)).data.createdBy == request.auth.uid
        && getAfter(/databases/$(database)/documents/groups/$(request.resource.data.groupId)).data.passcode == code;
      allow update, delete: if false;
    }

    match /groups/{groupId} {
      allow read: if isMember(groupId);
      allow create: if signedIn() && verified()
        && request.resource.data.createdBy == request.auth.uid
        && request.resource.data.penalty is number
        && request.resource.data.penalty >= 0
        && request.resource.data.penalty <= 100000
        && validCurrency(request.resource.data.get('currency', 'USD'));
      // Members may rename the group. Passcode, creator, timezone, the base
      // penalty and the base currency are fixed; changes go through the logs below.
      allow update: if isMember(groupId)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['name', 'photoRequired'])
        && request.resource.data.get('photoRequired', true) is bool;
      allow delete: if false;

      // Joining: you may add only yourself, and only with the group's real passcode.
      match /members/{uid} {
        allow read: if isMember(groupId);
        allow create: if signedIn() && verified()
          && request.auth.uid == uid
          && request.resource.data.joinedAt == request.time
          && (!('timezone' in request.resource.data) || validTz(request.resource.data.timezone))
          && getAfter(/databases/$(database)/documents/passcodes/$(request.resource.data.passcode)).data.groupId == groupId;
        // You may update your own name/photo, and Leave once (leftAt = server time).
        // Join time, timezone and leave time are locked afterwards.
        allow update: if signedIn() && request.auth.uid == uid
          && request.resource.data.joinedDate == resource.data.joinedDate
          && request.resource.data.joinedAt == resource.data.joinedAt
          && request.resource.data.get('timezone', '') == resource.data.get('timezone', '')
          && (request.resource.data.get('leftAt', null) == resource.data.get('leftAt', null)
              || (resource.data.get('leftAt', null) == null && request.resource.data.leftAt == request.time));
        allow delete: if false;
      }

      // Check-ins: whole group reads; you create only your own, stamped with server time.
      // If the group requires a photo, the photo must be written in the same batch.
      // You can undo your own check-in within 24 hours (the app offers it for today only).
      match /checkins/{checkinId} {
        allow read: if isMember(groupId);
        allow create: if isActiveMember(groupId)
          && request.resource.data.uid == request.auth.uid
          && checkinId == request.resource.data.date + '_' + request.auth.uid
          && request.resource.data.createdAt == request.time
          && (get(/databases/$(database)/documents/groups/$(groupId)).data.get('photoRequired', true) == false
              || existsAfter(/databases/$(database)/documents/groups/$(groupId)/photos/$(checkinId)));
        allow update: if false;
        allow delete: if isMember(groupId)
          && resource.data.uid == request.auth.uid
          && request.time < resource.data.createdAt + duration.value(1, 'd');
      }

      // Rest-day votes for a day: {votes: {uid: serverTime}}. You can add or
      // remove only your own vote; the vote is stamped with server time.
      match /restdays/{day} {
        allow read: if isMember(groupId);
        allow create: if isActiveMember(groupId)
          && request.resource.data.keys().hasOnly(['votes'])
          && request.resource.data.votes.keys().hasOnly([request.auth.uid])
          && request.resource.data.votes[request.auth.uid] == request.time;
        allow update: if isActiveMember(groupId)
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['votes'])
          && request.resource.data.votes.diff(resource.data.votes).affectedKeys().hasOnly([request.auth.uid])
          && (!(request.auth.uid in request.resource.data.votes)
              || request.resource.data.votes[request.auth.uid] == request.time);
        allow delete: if false;
      }

      // Workout photos: up to 4 per person per day (slot 1 is the one a check-in
      // needs). The image itself lives in Google Drive; this doc points to it.
      // Stamped with server time, never edited. Activity tag is optional.
      match /photos/{photoId} {
        allow read: if isMember(groupId);
        allow create: if isActiveMember(groupId)
          && request.resource.data.uid == request.auth.uid
          && photoId in [
            request.resource.data.date + '_' + request.auth.uid,
            request.resource.data.date + '_' + request.auth.uid + '_2',
            request.resource.data.date + '_' + request.auth.uid + '_3',
            request.resource.data.date + '_' + request.auth.uid + '_4'
          ]
          && request.resource.data.createdAt == request.time
          && request.resource.data.fileId is string
          && (!('activity' in request.resource.data)
              || request.resource.data.activity in ['Gym', 'Run', 'Tennis', 'Swim', 'Yoga', 'Cycle', 'Sport', 'Other']);
        allow update: if false;
        allow delete: if isMember(groupId)
          && resource.data.uid == request.auth.uid
          && request.time < resource.data.createdAt + duration.value(1, 'd');
      }

      // Weekly-target changes: permanent log. Each takes effect from a Monday.
      match /targets/{changeId} {
        allow read: if isMember(groupId);
        allow create: if isActiveMember(groupId)
          && request.resource.data.setBy == request.auth.uid
          && request.resource.data.setAt == request.time
          && request.resource.data.target is int
          && request.resource.data.target >= 1
          && request.resource.data.target <= 7
          && request.resource.data.effectiveDate is string;
        allow update, delete: if false;
      }

      // Penalty changes: permanent log. Each takes effect from effectiveDate.
      match /penalties/{changeId} {
        allow read: if isMember(groupId);
        allow create: if isActiveMember(groupId)
          && request.resource.data.setBy == request.auth.uid
          && request.resource.data.setAt == request.time
          && request.resource.data.penalty is number
          && request.resource.data.penalty >= 0
          && request.resource.data.penalty <= 100000
          && request.resource.data.effectiveDate is string;
        allow update, delete: if false;
      }

      // Timezone changes: permanent log. Only your own; the app sets them to
      // start from your next day (the ledger ignores any that don't).
      match /tzchanges/{changeId} {
        allow read: if isMember(groupId);
        allow create: if isMember(groupId)
          && request.resource.data.keys().hasOnly(['uid', 'timezone', 'effectiveDate', 'setAt'])
          && request.resource.data.uid == request.auth.uid
          && request.resource.data.setAt == request.time
          && validTz(request.resource.data.timezone)
          && request.resource.data.effectiveDate is string;
        allow update, delete: if false;
      }

      // Currency changes: permanent log, with the exchange rate on the day.
      match /currencies/{changeId} {
        allow read: if isMember(groupId);
        allow create: if isActiveMember(groupId)
          && request.resource.data.setBy == request.auth.uid
          && request.resource.data.setAt == request.time
          && validCurrency(request.resource.data.currency)
          && validCurrency(request.resource.data.from)
          && request.resource.data.currency != request.resource.data.from
          && request.resource.data.rate is number
          && request.resource.data.rate > 0
          && request.resource.data.penalty is number
          && request.resource.data.penalty >= 0
          && request.resource.data.penalty <= 100000
          && request.resource.data.effectiveDate is string;
        allow update, delete: if false;
      }
    }

    // Private health data (meals, weight): owner only. Used in later phases.
    match /private/{uid}/{document=**} {
      allow read, write: if signedIn() && request.auth.uid == uid;
    }
  }
}
